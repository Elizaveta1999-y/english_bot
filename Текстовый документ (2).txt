import asyncio
import os
from dotenv import load_dotenv

# Загружаем переменные из файла .env (сразу после импорта)
load_dotenv()

from aiogram import Bot, Dispatcher

# Импортируем все хендлеры, которые есть в проекте
from handlers import start, speaking, roleplay, common, voice, lessons, words, profile, support, listening, reading, writing
from handlers.grammar import router as grammar_router
from handlers.govorenie import router as govorenie_router
# Если есть другие роутеры (например, subscription), добавь их ниже
# from handlers.subscription import router as subscription_router

from utils.db import init_db

# Получаем токен из переменных окружения
BOT_TOKEN = os.getenv("BOT_TOKEN")
if not BOT_TOKEN:
    raise ValueError("BOT_TOKEN не найден в .env файле")

bot = Bot(token=BOT_TOKEN)
dp = Dispatcher()

# Подключаем все роутеры (так же, как в app.py, но без вебхуков)
dp.include_router(start.router)
dp.include_router(words.router)
dp.include_router(govorenie_router)
dp.include_router(writing.router)
dp.include_router(reading.router)
dp.include_router(listening.router)
dp.include_router(grammar_router)
dp.include_router(support.router)
# dp.include_router(subscription_router)  # если есть, раскомментируй
dp.include_router(roleplay.router)
dp.include_router(voice.router)
dp.include_router(common.router)
dp.include_router(lessons.router)
dp.include_router(speaking.router)
dp.include_router(profile.router)

async def main():
    # Инициализация базы данных (создаёт таблицы)
    await init_db()
    print("🚀 Бот запущен локально (polling)")
    print("Нажми Ctrl+C для остановки")
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
